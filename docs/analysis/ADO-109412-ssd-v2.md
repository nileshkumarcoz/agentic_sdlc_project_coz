```

│   ├── __init__.py            # Makes module importable

│   ├── audit_logger.py        # NEW: structured JSON audit logging via RotatingFileHandler
```

**Component Responsibilities:**

| Component | Responsibility |
|---|---|

| `main.py` | Orchestrates user prompts, calls calculator, calls audit logger, prints results/errors. |

### 2. Data Model

#### 2.1 In-Memory Flow

```
raw_input_1: str  ──► validate_input() ──► num1: float ──┐
raw_input_2: str  ──► validate_input() ──► num2: float ──┤
                                                          ▼

                                                          │

                                                          │
                                               display_result() → stdout
```

#### 2.2 Audit Log Entry Schema (JSON Lines)

Each line in `logs/audit.log` is a self-contained JSON object:

```json
// SUCCESS
{
  "timestamp": "2024-07-15T10:23:45.123456Z",
  "event": "SUCCESS",
  "input_1": "42",
  "input_2": "8",
  "result": 50.0,
  "error": null,
  "hostname": "dev-machine",
  "os_user": "sushil"
}

// VALIDATION_ERROR
{
  "timestamp": "2024-07-15T10:24:01.654321Z",
  "event": "VALIDATION_ERROR",
  "input_1": "abc",
  "input_2": "8",
  "result": null,
  "error": "'abc' is not a valid numeric value.",
  "hostname": "dev-machine",
  "os_user": "sushil"
}
```

**Fields:**

| Field | Type | Description |
|---|---|---|
| `timestamp` | `str` (ISO-8601 UTC) | When the event occurred |
| `event` | `str` | `SUCCESS` or `VALIDATION_ERROR` |
| `input_1` | `str` | Raw first user input |
| `input_2` | `str` | Raw second user input |
| `result` | `float \| null` | Computed sum on success; null on error |
| `error` | `str \| null` | Error description on failure; null on success |
| `hostname` | `str` | Machine hostname for traceability |
| `os_user` | `str` | OS-level username for accountability |

### 3. Key APIs / Interfaces

#### 3.1 `calculator.py`

```python
def validate_input(value: str) -> float:
    """
    Validates that the input string is non-empty and numeric.

    Args:
        value (str): Raw string input from the user.

    Returns:
        float: The numeric representation of the input.

    Raises:
        ValueError: If the input is empty or non-numeric.
    """
    if not value.strip():
        raise ValueError("Input must not be empty.")
    try:
        return float(value.strip())
    except ValueError:
        raise ValueError(f"'{value.strip()}' is not a valid numeric value.")


def calculate_sum(a: float, b: float) -> float:
    """
    Returns the arithmetic sum of two floats.

    Args:
        a (float): First number.
        b (float): Second number.

    Returns:
        float: a + b
    """
    return a + b
```

#### 3.2 `audit_logger.py` *(NEW)*

```python
import logging
import json
import os
import socket
import getpass
from datetime import datetime, timezone
from logging.handlers import RotatingFileHandler
from typing import Optional


def _get_audit_logger() -> logging.Logger:
    """
    Builds and returns a logger writing JSON lines to logs/audit.log.
    Uses RotatingFileHandler: max 1 MB per file, 5 backups.
    Safe to call multiple times (idempotent via logger name check).
    """
    logger_name = "add_two_numbers.audit"
    logger = logging.getLogger(logger_name)

    if not logger.handlers:  # Prevent duplicate handlers on re-import
        log_dir = os.environ.get("AUDIT_LOG_DIR", "logs")
        os.makedirs(log_dir, exist_ok=True)
        log_path = os.path.join(log_dir, "audit.log")

        handler = RotatingFileHandler(
            log_path, maxBytes=1_048_576, backupCount=5, encoding="utf-8"
        )
        handler.setFormatter(logging.Formatter("%(message)s"))
        logger.addHandler(handler)
        logger.setLevel(logging.INFO)
        logger.propagate = False  # Do not bubble up to root logger

    return logger


def _build_entry(
    event: str,
    input_1: str,
    input_2: str,
    result: Optional[float],
    error: Optional[str],
) -> str:
    """Serialises an audit entry to a JSON string."""
    return json.dumps({
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "event": event,
        "input_1": input_1,
        "input_2": input_2,
        "result": result,
        "error": error,
        "hostname": socket.gethostname(),
        "os_user": _safe_getuser(),
    })


def _safe_getuser() -> str:
    """Returns OS username; falls back gracefully in restricted environments."""
    try:
        return getpass.getuser()
    except Exception:
        return "unknown"


def log_success(input_1: str, input_2: str, result: float) -> None:
    """
    Writes a SUCCESS audit entry.

    Args:
        input_1 (str): Raw first input string.
        input_2 (str): Raw second input string.
        result (float): Computed sum.
    """
    try:
        _get_audit_logger().info(_build_entry("SUCCESS", input_1, input_2, result, None))
    except Exception:
        pass  # Never crash the main program due to logging failure


def log_failure(input_1: str, input_2: str, error: str) -> None:
    """
    Writes a VALIDATION_ERROR audit entry.

    Args:
        input_1 (str): Raw first input string.
        input_2 (str): Raw second input string.
        error (str): Description of the validation error.
    """
    try:
        _get_audit_logger().info(_build_entry("VALIDATION_ERROR", input_1, input_2, None, error))
    except Exception:
        pass  # Never crash the main program due to logging failure
```

#### 3.3 `main.py`

```python
from add_two_numbers.calculator import validate_input, calculate_sum
from add_two_numbers import audit_logger


def main() -> None:
    """CLI entry point orchestrating the full add-two-numbers flow with audit logging."""
    print("=== Add Two Numbers ===")
    raw1 = input("Enter the first number: ")
    raw2 = input("Enter the second number: ")

    try:
        num1 = validate_input(raw1)
        num2 = validate_input(raw2)
        result = calculate_sum(num1, num2)
        print(f"Result: {num1} + {num2} = {result}")
        audit_logger.log_success(raw1, raw2, result)

    except ValueError as e:
        print(f"Error: {e}")
        audit_logger.log_failure(raw1, raw2, str(e))


if __name__ == "__main__":
    main()
```

### 5. Technology Choices

| Concern | Choice | Rationale |
|---|---|---|
| Language | Python 3.8+ | Specified in story; zero external runtime deps |
| Type conversion | Built-in `float()` | Handles int and float strings natively |
| Error handling | `try/except ValueError` | Standard Pythonic pattern; prevents crashes |
| **Audit logging** | **Python `logging` + `RotatingFileHandler`** | **Standard library; no extra dependencies; built-in rotation; thread-safe writes** |
| **Log format** | **JSON Lines (one JSON object per line)** | **Machine-parseable; easy to ingest into Splunk/CloudWatch later; human-readable** |
| **Log path config** | **`AUDIT_LOG_DIR` env var (default: `logs/`)** | **12-Factor App compliant; allows override in CI/CD without code changes** |
| Testing framework | `pytest` + `pytest-cov` | Industry standard; rich assertion introspection; coverage reports |
| Code style | PEP 8 + `flake8` (dev) | Enforces maintainability NFR |

### 6. Non-Functional Design

#### 6.1 Security
- No `eval()` usage — prevents code injection from user input.
- All inputs sanitised through `float()` inside `try/except`.
- Audit log records only numeric strings — no sensitive PII expected.
- Log directory permissions should be set to `700` in production deployments.

#### 6.2 Scalability
- O(1) arithmetic; stateless functions.
- `RotatingFileHandler` caps log growth at 5 MB (1 MB × 5 backups) automatically.
- Log path is environment-variable-configurable for containerised deployments.

#### 6.3 Observability *(Enhanced)*
- **Audit log** (`logs/audit.log`): JSON-lines format, one entry per calculation attempt.
- Every entry carries `timestamp`, `event`, `hostname`, `os_user`, inputs, result/error.
- Future: forward `audit.log` to a log aggregator (ELK, Splunk, AWS CloudWatch) by pointing `AUDIT_LOG_DIR` at a watched directory or replacing the handler.

#### 6.4 Resilience of Logging
- All `audit_logger` calls are wrapped in `try/except Exception: pass`.
- A logging failure **never propagates** to the main program — the user still gets their result.
- A `PermissionError` on log directory creation will be silently swallowed at the logger level.

#### 6.5 Maintainability
- Pure `calculator.py` functions have zero side-effects — independently testable.
- `audit_logger.py` is a standalone module — replaceable with a remote logging backend without touching `calculator.py` or `main.py`.
- `main()` is separated from `__main__` guard to allow import-time testing.

#### 7.2 `tests/test_audit_logger.py` — Audit Log Unit Tests *(NEW)*

| Test ID | Scenario | Expected Log Entry |
|---|---|---|
| AL-01 | `log_success()` writes correct JSON | `event == "SUCCESS"`, `result == 50.0`, `error == null` |
| AL-02 | `log_failure()` writes correct JSON | `event == "VALIDATION_ERROR"`, `result == null`, `error` is non-empty string |
| AL-03 | Entry contains `timestamp` in ISO-8601 format | `timestamp` field parseable by `datetime.fromisoformat()` |
| AL-04 | Entry contains `hostname` and `os_user` | Both fields are non-empty strings |
| AL-05 | Log does not raise on permission error | Patch `RotatingFileHandler` to raise; assert no exception propagates |
| AL-06 | `AUDIT_LOG_DIR` env var is respected | Set env var to temp dir; assert log written there |

```python
import json
import os
import pytest
from unittest.mock import patch
from datetime import datetime


@pytest.fixture(autouse=True)
def isolated_log_dir(tmp_path, monkeypatch):
    """Redirect audit logs to a temp directory for each test."""
    monkeypatch.setenv("AUDIT_LOG_DIR", str(tmp_path / "logs"))
    # Clear cached logger handlers between tests
    import logging
    logger = logging.getLogger("add_two_numbers.audit")
    logger.handlers.clear()
    yield tmp_path / "logs"


def read_last_log_entry(log_dir) -> dict:
    log_path = os.path.join(log_dir, "audit.log")
    with open(log_path, "r") as f:
        lines = [l.strip() for l in f.readlines() if l.strip()]
    return json.loads(lines[-1])


class TestAuditLoggerSuccess:
    def test_success_event_type(self, isolated_log_dir):
        from add_two_numbers.audit_logger import log_success
        log_success("42", "8", 50.0)
        entry = read_last_log_entry(isolated_log_dir)
        assert entry["event"] == "SUCCESS"

    def test_success_result_value(self, isolated_log_dir):
        from add_two_numbers.audit_logger import log_success
        log_success("42", "8", 50.0)
        entry = read_last_log_entry(isolated_log_dir)
        assert entry["result"] == 50.0
        assert entry["error"] is None

    def test_success_contains_timestamp(self, isolated_log_dir):
        from add_two_numbers.audit_logger import log_success
        log_success("1", "2", 3.0)
        entry = read_last_log_entry(isolated_log_dir)
        datetime.fromisoformat(entry["timestamp"])  # Must not raise

    def test_success_contains_host_and_user(self, isolated_log_dir):
        from add_two_numbers.audit_logger import log_success
        log_success("1", "2", 3.0)
        entry = read_last_log_entry(isolated_log_dir)
        assert isinstance(entry["hostname"], str) and entry["hostname"]
        assert isinstance(entry["os_user"], str) and entry["os_user"]


class TestAuditLoggerFailure:
    def test_failure_event_type(self, isolated_log_dir):
        from add_two_numbers.audit_logger import log_failure
        log_failure("abc", "8", "'abc' is not a valid numeric value.")
        entry = read_last_log_entry(isolated_log_dir)
        assert entry["event"] == "VALIDATION_ERROR"

    def test_failure_null_result(self, isolated_log_dir):
        from add_two_numbers.audit_logger import log_failure
        log_failure("abc", "8", "some error")
        entry = read_last_log_entry(isolated_log_dir)
        assert entry["result"] is None
        assert entry["error"] == "some error"


class TestAuditLoggerResilience:
    def test_log_failure_does_not_propagate_exception(self, monkeypatch):
        """Logging errors must never crash the main program."""
        import add_two_numbers.audit_logger as al
        monkeypatch.setattr(al, "_get_audit_logger", lambda: (_ for _ in ()).throw(OSError("disk full")))
        # Must not raise
        al.log_success("1", "2", 3.0)
        al.log_failure("x", "2", "error")
```

#### 7.3 `tests/test_main.py` — Integration Tests

```python
import pytest


def test_main_happy_path_prints_result(capsys, tmp_path, monkeypatch):

def test_main_happy_path_writes_success_audit_log(tmp_path, monkeypatch):

    monkeypatch.setenv("AUDIT_LOG_DIR", str(tmp_path / "logs"))
```
